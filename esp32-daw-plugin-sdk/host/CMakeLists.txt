# The native test host, linked with every plugin of this SDK through a generated registry.
get_property(_names GLOBAL PROPERTY DAWSDK_PLUGINS)
get_property(_entries GLOBAL PROPERTY DAWSDK_PLUGIN_ENTRIES)
set(_decl "")
set(_list "")
foreach(e ${_entries})
  string(APPEND _decl "extern \"C\" const daw_plugin_descriptor_t *${e}(void);\n")
  string(APPEND _list "        ${e}(),\n")
endforeach()
set(_reg ${CMAKE_CURRENT_BINARY_DIR}/plugin_registry.cpp)
file(CONFIGURE OUTPUT ${_reg} CONTENT "// generated by host/CMakeLists.txt
#include \"dawsdk_host/host.hpp\"
${_decl}
namespace dawhost {
const std::vector<const daw_plugin_descriptor_t *> &registry()
{
    static const std::vector<const daw_plugin_descriptor_t *> r = {
${_list}    };
    return r;
}
}
")
add_library(dawsdk_host STATIC host.cpp modgen.cpp ${_reg})
target_include_directories(dawsdk_host PUBLIC include)
target_link_libraries(dawsdk_host PUBLIC dawsdk_api)
foreach(n ${_names})
  target_link_libraries(dawsdk_host PUBLIC dawplug_${n})
endforeach()
target_compile_options(dawsdk_host PRIVATE -Wall -Wextra)
